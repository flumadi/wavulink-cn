! LTE-GW  (Cisco 1941)  - simulates the 4G backup operator (carrier NAT)
! G0/0 -> HUB-CORE G0/2   G0/1 -> INTERNET-SW
enable
configure terminal
hostname LTE-GW
no ip domain-lookup
interface GigabitEthernet0/0
 description TO-LOKORI-HUB
 ip address 100.65.0.1 255.255.255.252
 ip nat inside
 no shutdown
interface GigabitEthernet0/1
 description INTERNET
 ip address 203.0.113.2 255.255.255.0
 ip nat outside
 no shutdown
ip route 10.10.0.0 255.255.0.0 100.65.0.2
access-list 1 permit 10.10.0.0 0.0.255.255
ip nat inside source list 1 interface GigabitEthernet0/1 overload
end
